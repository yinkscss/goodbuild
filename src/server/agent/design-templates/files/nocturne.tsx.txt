import { useEffect, useRef, useState } from "react";
import { templateImage } from "./lib/templateImage";
import { content } from "./lib/templateContent";
import "./showcase.css";

const presets = {
  Studio: [26, 39, 48, 56, 69, 80, 73, 60, 46, 35, 28, 20],
  Focus: [15, 20, 30, 48, 73, 95, 95, 73, 48, 30, 20, 15],
  Night: [65, 80, 94, 82, 67, 49, 36, 25, 18, 14, 10, 7],
};
export default function App() {
  const [preset, setPreset] = useState<keyof typeof presets>("Studio");
  const [audioNote, setAudioNote] = useState("");
  const [playing, setPlaying] = useState(false);
  const audio = useRef<AudioContext | null>(null);
  useEffect(() => () => { if (audio.current && audio.current.state !== "closed") void audio.current.close().catch(() => undefined); }, []);
  async function play() {
    if (playing) return;
    if (!window.AudioContext) { setAudioNote("Your browser doesn't support the test tone. The preset visualizer still works."); return; }
    try {
      await audio.current?.close();
      const context = new window.AudioContext(); audio.current = context;
      await context.resume();
      const oscillator = context.createOscillator(); const gain = context.createGain();
      oscillator.type = "sine"; oscillator.frequency.value = preset === "Night" ? 220 : preset === "Focus" ? 440 : 330;
      gain.gain.setValueAtTime(0, context.currentTime); gain.gain.linearRampToValueAtTime(.025, context.currentTime + .1); gain.gain.linearRampToValueAtTime(0, context.currentTime + 1.3);
      oscillator.connect(gain); gain.connect(context.destination); setPlaying(true); setAudioNote("Playing a quiet test tone for 1.4 seconds.");
      oscillator.onended = () => { setPlaying(false); setAudioNote("Test tone finished."); if (context.state !== "closed") void context.close().catch(() => undefined); if (audio.current === context) audio.current = null; };
      oscillator.start(); oscillator.stop(context.currentTime + 1.4);
    } catch { setPlaying(false); setAudioNote("Audio couldn't start. Try again after allowing sound in your browser."); }
  }
  return <div className="showcase nocturne">
    <a className="skip-link" href="#sound">Skip to sound controls</a>
    <header className="site-nav"><a className="nocturne-brand" href="#"><span aria-hidden="true">◒</span>{content.brand.toLowerCase()}</a><nav className="nav-links" aria-label="Primary"><a href="#sound">The sound</a><a className="nav-optional" href="#details">The details</a><a className="quiet-button" href="#sound">Explore N° 01 <span aria-hidden="true">↗</span></a></nav></header>
    <main>
      <section className="nocturne-hero" aria-labelledby="nocturne-title"><img className="nocturne-product" src={templateImage.src} alt={templateImage.alt} width={1200} height={675} fetchPriority="high" /><div className="nocturne-copy enter"><p className="nocturne-edition"><span /> Introducing N° 01 / Concept edition</p><h1 id="nocturne-title">{content.headline}</h1><p className="nocturne-sub">{content.sub}</p><a className="solid-button" href="#sound">Find your frequency <span aria-hidden="true">↗</span></a></div><div className="nocturne-side"><span>A little closer to the music.</span><div className="nocturne-orbit" aria-hidden="true"><i /><i /><i /></div></div><div className="nocturne-bottom"><p>Precision in every detail.<br /><span>Presence in every note.</span></p><div className="nocturne-wave" aria-hidden="true">{Array.from({ length: 49 }, (_, i) => <i key={i} style={{ height: `${8 + ((i * 17 + i * i) % 36)}px`, animationDelay: `${i * -.06}s` }} />)}</div><span className="fine">Designed to disappear.<br />Made to feel everything.</span></div></section>
      <section className="nocturne-sound" id="sound" aria-labelledby="sound-title"><div><p className="fine">One listening experience. Your way.</p><h2 id="sound-title" className="section-heading">Set the mood.<br />Keep the detail.</h2><p>Shape a different listening moment. Explore the visual presets, then tap to hear a quiet test tone.</p><div className="nocturne-presets" role="group" aria-label="Sound preset">{(Object.keys(presets) as Array<keyof typeof presets>).map((name) => <button key={name} aria-pressed={preset === name} onClick={() => setPreset(name)}>{name}</button>)}</div><button className="quiet-button" onClick={play} disabled={playing}>{playing ? "Playing test tone…" : "Play test tone"}</button><p className="audio-status" role="status">{audioNote}</p></div><div className="nocturne-eq" data-preset={preset} role="img" aria-label={`${preset} equalizer preset visualization`}><div className="eq-header"><span>{preset} mode</span><span>Concept sound profile</span></div><div className="eq-bars" aria-hidden="true">{presets[preset].map((height, i) => <i key={i} style={{ height: `${height}%` }} />)}</div><div className="eq-scale"><span>Low</span><span>Mid</span><span>High</span></div></div></section>
      <section className="nocturne-details" id="details"><h2>A considered kind of quiet.</h2><div><p>A sculptural frame. Tactile materials. A shape that feels at home with you.</p><p className="fine muted">This is an original product concept. No hardware performance, battery life or noise-cancellation claim is being made. Presets change the visualization and test-tone pitch; they do not alter your device's sound settings.</p></div></section>
    </main><footer className="site-footer"><span>{content.brand} / Make room for listening.</span><span>Original concept. No checkout or payment.</span></footer>
  </div>;
}
